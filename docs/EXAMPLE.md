# Example: a `user` feature

A complete feature built the way this starter expects: create, get and list users, with
validation, a business rule (unique email), response filtering (`passwordHash` never leaves
the server), pagination and a stricter rate limit on signup.

Storage is an in-memory adapter so the example runs as-is. Replacing it with Prisma or
TypeORM means writing one new class that implements the same port — no use-case, DTO or
controller changes.

Every file below was compiled, linted and exercised with real requests against a project
generated by the CLI.

## 1. Use-case layer

Business types, the port the use-cases depend on, and one class per operation.
Nothing here knows about HTTP, Express or a database.

```ts
/**
 * src/use-case/user/model/User.ts
 */

export type User = {
  id: string;
  email: string;
  name: string;
  passwordHash: string;
  createdAt: Date;
};
```

```ts
/**
 * src/use-case/user/port/UserRepositoryPort.ts
 */
import type { User } from '../model/User';

export const USER_REPOSITORY = Symbol('USER_REPOSITORY');

export interface UserRepositoryPort {
  findById(id: string): Promise<User | null>;
  findByEmail(email: string): Promise<User | null>;
  findMany(offset: number, limit: number): Promise<[User[], number]>;
  create(data: Omit<User, 'id' | 'createdAt'>): Promise<User>;
}
```

```ts
/**
 * src/use-case/user/CreateUserUseCase.ts
 */
import { Inject, Injectable } from '@nestjs/common';

import { createHash } from 'node:crypto';

import { EmailTakenError } from '@/shared/error';

import type { User } from './model/User';
import {
  USER_REPOSITORY,
  type UserRepositoryPort,
} from './port/UserRepositoryPort';

export type CreateUserInput = { email: string; name: string; password: string };

@Injectable()
export class CreateUserUseCase {
  constructor(
    @Inject(USER_REPOSITORY) private readonly users: UserRepositoryPort,
  ) {}

  async execute(input: CreateUserInput): Promise<User> {
    if (await this.users.findByEmail(input.email)) {
      throw new EmailTakenError(`Email ${input.email} is already registered`);
    }

    return this.users.create({
      email: input.email,
      name: input.name,
      /**
       * Example only: use argon2 or bcrypt for real passwords
       */
      passwordHash: createHash('sha256').update(input.password).digest('hex'),
    });
  }
}
```

```ts
/**
 * src/use-case/user/GetUserUseCase.ts
 */
import { Inject, Injectable } from '@nestjs/common';

import { UserNotFoundError } from '@/shared/error';

import type { User } from './model/User';
import {
  USER_REPOSITORY,
  type UserRepositoryPort,
} from './port/UserRepositoryPort';

@Injectable()
export class GetUserUseCase {
  constructor(
    @Inject(USER_REPOSITORY) private readonly users: UserRepositoryPort,
  ) {}

  async execute(id: string): Promise<User> {
    const user = await this.users.findById(id);
    if (!user) throw new UserNotFoundError(`User ${id} not found`);

    return user;
  }
}
```

```ts

}
```

## 3. Infrastructure adapter

Implements the port. Swap this class for `PrismaUserRepository` later.

```ts
/**
 * src/infrastructure/database/repository/InMemoryUserRepository.ts
 */
import { Injectable } from '@nestjs/common';

import { randomUUID } from 'node:crypto';

import type { User, UserRepositoryPort } from '@/use-case/user';

@Injectable()
export class InMemoryUserRepository implements UserRepositoryPort {
  private readonly users = new Map<string, User>();

  findById(id: string): Promise<User | null> {
    return Promise.resolve(this.users.get(id) ?? null);
  }

  findByEmail(email: string): Promise<User | null> {
    const user = [...this.users.values()].find((u) => u.email === email);
    return Promise.resolve(user ?? null);
  }

  findMany(offset: number, limit: number): Promise<[User[], number]> {
    const all = [...this.users.values()];
    return Promise.resolve([all.slice(offset, offset + limit), all.length]);
  }

  create(data: Omit<User, 'id' | 'createdAt'>): Promise<User> {
    const user: User = { ...data, id: randomUUID(), createdAt: new Date() };
    this.users.set(user.id, user);
    return Promise.resolve(user);
  }
}

/**
 * src/infrastructure/database/repository/index.ts
 */
export * from './InMemoryUserRepository';
```

## 4. Presentation: DTOs

Request DTOs validate input. The response DTO decides which fields are sent:
`passwordHash` is not in the schema, so it is removed from every response.

```ts
/**
 * src/presentation/dto/user/request/CreateUserRequestDto.ts
 */
import { createZodDto } from 'nestjs-zod';
import { z } from 'zod';

export const CreateUserRequestSchema = z.object({
  email: z.email(),
  name: z.string().trim().min(2).max(100),
  password: z.string().min(8).max(128),
});

export class CreateUserRequestDto extends createZodDto(
  CreateUserRequestSchema,
) {}
```

```ts
/**
 * src/presentation/dto/user/request/GetUserRequestDto.ts — route params are validated too
 */
import { createZodDto } from 'nestjs-zod';
import { z } from 'zod';

export const GetUserRequestSchema = z.object({ id: z.uuid() });

export class GetUserRequestDto extends createZodDto(GetUserRequestSchema) {}
```

```ts
/**
 * src/presentation/dto/user/response/UserResponseDto.ts
 */
import { createZodDto } from 'nestjs-zod';
import { z } from 'zod';

/**
 * Date in code, ISO string in JSON and in the OpenAPI document
 */
const isoDate = z.codec(z.iso.datetime(), z.date(), {
  decode: (iso) => new Date(iso),
  encode: (date) => date.toISOString(),
});

export const UserResponseSchema = z.object({
  id: z.uuid(),
  email: z.email(),
  name: z.string(),
  createdAt: isoDate,
});

export class UserResponseDto extends createZodDto(UserResponseSchema, {
  codec: true,
}) {}
```

> **Dates:** a plain `z.date()` in a response DTO crashes the app at startup
> (`Date cannot be represented in JSON Schema`). Use a codec as above and
> `createZodDto(schema, { codec: true })`.

```ts
/**
 * src/presentation/dto/user/index.ts
 */

export * from './request/CreateUserRequestDto';
export * from './request/GetUserRequestDto';
export * from './response/UserResponseDto';
```

## 5. Presentation: controller

Thin: validated input in, one use-case call, plain data out. The envelope, serialization,
errors and Swagger docs all come from the global setup and `@ApiEnvelopeResponse`.

```ts
/**
 * src/presentation/controller/user/UserController.ts
 */
import {
  Body,
  Controller,
  Get,
  HttpStatus,
  Param,
  Post,
  Query,
} from '@nestjs/common';
import { ApiTags } from '@nestjs/swagger';

import {
  CreateUserRequestDto,
  GetUserRequestDto,
  UserResponseDto,
} from '@/presentation/dto/user';

import {
  CreateUserUseCase,
  GetUserUseCase,
  ListUsersUseCase,
} from '@/use-case/user';

import { ApiEnvelopeResponse, RateLimit } from '@/shared/decorator';
import { PageDto, PaginationQueryDto } from '@/shared/dto';

@ApiTags('users')
@Controller('users')
export class UserController {
  constructor(
    private readonly createUser: CreateUserUseCase,
    private readonly getUser: GetUserUseCase,
    private readonly listUsers: ListUsersUseCase,
  ) {}

  @Post()
  /**
   * signup: 5 per minute per IP
   */
  @RateLimit(5, 60)
  @ApiEnvelopeResponse(UserResponseDto, { status: HttpStatus.CREATED })
  create(@Body() body: CreateUserRequestDto) {
    return this.createUser.execute(body);
  }

  @Get()
  @ApiEnvelopeResponse(UserResponseDto, { paginated: true })
  async list(@Query() query: PaginationQueryDto) {
    const [users, total] = await this.listUsers.execute(
      query.offset,
      query.limit,
    );
    return PageDto.of(users, total, query);
  }

  @Get(':id')
  @ApiEnvelopeResponse(UserResponseDto)
  get(@Param() { id }: GetUserRequestDto) {
    return this.getUser.execute(id);
  }
}

/**
 * src/presentation/controller/user/index.ts
 */
export * from './UserController';
```

## 6. App: wire it together

The feature module binds the port to the adapter; `AppModule` imports the feature module.

```ts
/**
 * src/app/user/UserModule.ts
 */
import { Module } from '@nestjs/common';

import { UserController } from '@/presentation/controller/user';

import { InMemoryUserRepository } from '@/infrastructure/database/repository';

import {
  CreateUserUseCase,
  GetUserUseCase,
  ListUsersUseCase,
  USER_REPOSITORY,
} from '@/use-case/user';

@Module({
  controllers: [UserController],
  providers: [
    CreateUserUseCase,
    GetUserUseCase,
    ListUsersUseCase,
    { provide: USER_REPOSITORY, useClass: InMemoryUserRepository },
  ],
})
export class UserModule {}

/**
 * src/app/user/index.ts

export * from './UserModule';
```

## 8. Test a use-case

Use-cases need no Nest app and no database — pass a fake port:

```ts
/**
 * src/use-case/user/GetUserUseCase.spec.ts
 */
import { describe, expect, it } from 'bun:test';

import { UserNotFoundError } from '@/shared/error';

import { GetUserUseCase } from './GetUserUseCase';
import type { UserRepositoryPort } from './port/UserRepositoryPort';

const emptyRepo = {
  findById: () => Promise.resolve(null),
} as unknown as UserRepositoryPort;

describe('GetUserUseCase', () => {
  it('throws UserNotFoundError for an unknown id', async () => {
    const useCase = new GetUserUseCase(emptyRepo);

    const error = await useCase.execute('missing').catch((e: unknown) => e);

    expect(error).toBeInstanceOf(UserNotFoundError);
  });
});
```

(`new` is fine in tests: the "never instantiate injectables" rule is about app code.)
